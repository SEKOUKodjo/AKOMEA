shared Parametres = [
    Dossier = "C:\RAPPORT_ISTS\CSV",

shared Transport = let
    P = Parametres,
    Dossier = P[Dossier] & "\TRANSPORT",
    Fichiers = if P[FichierTransport] <> null then null else Table.Sort(Table.SelectRows(Folder.Files(Dossier), each Text.Lower([Extension]) = ".csv"), {{"Date modified", Order.Descending}}),
    Binaire = if P[FichierTransport] <> null then File.Contents(P[FichierTransport]) else if Table.IsEmpty(Fichiers) then error Error.Record("Fichier CSV manquant", "Aucun fichier .csv dans le dossier " & Dossier) else Fichiers{0}[Content],
    Source = Csv.Document(Binaire, [Delimiter = P[Separateur], Encoding = 65001, QuoteStyle = QuoteStyle.Csv]),
    Entetes = Table.PromoteHeaders(Source, [PromoteAllScalars = true]),
    Colonnes = Table.SelectColumns(Entetes, {"region", "health_district", "health_facility", "site", "date", "vl_samples", "eid_samples", "tb_samples", "vl_results", "eid_results", "tb_results", "trips", "other_item_name", "other_item_qty", "duration_from_collection_days", "duration_in_hub1_days", "duration_in_hub2_days", "duration_reception_to_pickup_days", "action", "custom_month"}, MissingField.UseNull),
    Vides = Table.ReplaceValue(Colonnes, "", null, Replacer.ReplaceValue, Table.ColumnNames(Colonnes)),
    Textes = Table.TransformColumns(Vides, {{"region", Text.Trim, type text}, {"health_district", Text.Trim, type text}, {"health_facility", Text.Trim, type text}, {"site", Text.Trim, type text}, {"action", Text.Trim, type text}, {"custom_month", Text.Trim, type text}, {"other_item_name", Text.Trim, type text}}),
    Nombres = Table.TransformColumnTypes(Textes, List.Transform({"vl_samples", "eid_samples", "tb_samples", "vl_results", "eid_results", "tb_results", "trips", "other_item_qty", "duration_from_collection_days", "duration_in_hub1_days", "duration_in_hub2_days", "duration_reception_to_pickup_days"}, each {_, type number}), "en-US"),
    Dates = Table.TransformColumns(Nombres, {{"date", each if _ = null then null else try Date.FromText(Text.Start(Text.From(_), 10), "en-US") otherwise null, type date}}),
    AvecPeriode = Table.AddColumn(Dates, "Periode", each
        let
            cm = if [custom_month] = null then "" else [custom_month],
            fin = if Text.Length(cm) >= 10 then (try Date.FromText(Text.End(cm, 10), "en-US") otherwise null) else null,
            ref = if fin <> null then fin else if [date] = null then null else if Date.Day([date]) > 25 then Date.AddMonths([date], 1) else [date]
        in
            if ref = null then null else Date.StartOfMonth(ref), type date),
    Final = Table.SelectColumns(AvecPeriode, {"Periode", "region", "health_district", "health_facility", "site", "date", "vl_samples", "eid_samples", "tb_samples", "vl_results", "eid_results", "tb_results", "trips", "other_item_name", "other_item_qty", "duration_from_collection_days", "duration_in_hub1_days", "duration_in_hub2_days", "duration_reception_to_pickup_days", "action", "custom_month"})
in
    Final;

shared Rejets_FOSA = let
    P = Parametres,
    Raisons = {{"reason_haemolysed", "Haemolysed samples"}, {"reason_insufficient_volume", "Insufficient sample volume"}, {"reason_poorly_conserved", "Poorly conserved samples"}, {"reason_clotted", "Clotted samples"}, {"reason_illegible_label", "Illegible label"}, {"reason_empty_spilled", "Empty sample tubes/spilled sample"}, {"reason_no_label", "Sample lacking an identification (MPI) code / no label"}, {"reason_prolonged_storage", "Prolonged storage"}, {"reason_no_request_form", "No request form"}, {"reason_analytical_error", "Analytical / Manipulation error"}, {"reason_poor_packaging", "Poor packaging"}, {"reason_unmatched_info", "Unmatched information"}, {"reason_wrong_code", "Wrong code"}, {"reason_wrong_form", "Wrong form"}, {"reason_other_reason", "Other reason"}},
    Mois = if P[MoisRapport] <> null then P[MoisRapport] else List.Max(Rejets[Periode]),
    F = Table.SelectRows(Rejets, each [Periode] = Mois),
    G = Table.Group(F, {"region", "health_district", "health_facility", "site"}, {
        {"VL samples", each List.Sum([vl_rejected]), type nullable number},
        {"EID samples", each List.Sum([eid_rejected]), type nullable number},
        {"TB samples", each List.Sum([tb_rejected]), type nullable number},
        {"Total", each List.Sum([total_rejected]), type nullable number},
        {"Reasons for rejection", (t) => Text.Combine(List.RemoveNulls(List.Transform(Raisons, (r) => let n = List.Sum(Table.Column(t, r{0})) in if n = null or n = 0 then null else r{1} & " (" & Text.From(n) & ")")), "; "), type text}}),
    Tri = Table.Sort(G, {{"region", Order.Ascending}, {"Total", Order.Descending}}),
    Final = Table.RenameColumns(Tri, {{"region", "Region"}, {"health_district", "District"}, {"health_facility", "Health facility"}, {"site", "Site"}})
in
    Final;

shared Intrants_FOSA = let
    P = Parametres,
    Mois = if P[MoisRapport] <> null then P[MoisRapport] else List.Max(Transport[Periode]),
    F = Table.SelectRows(Transport, each [Periode] = Mois and [other_item_name] <> null and [other_item_qty] <> null and [other_item_qty] <> 0),
    G = Table.Group(F, {"region", "health_district", "health_facility", "other_item_name"}, {{"qty", each List.Sum([other_item_qty]), type number}}),
    Articles = List.Sort(List.Distinct(G[other_item_name])),
    Pivot = Table.Pivot(G, Articles, "other_item_name", "qty", List.Sum),
    AvecTotal = Table.AddColumn(Pivot, "Total", each List.Sum(List.RemoveNulls(Record.ToList(Record.SelectFields(_, Articles)))), type number),
    Tri = Table.Sort(AvecTotal, {{"region", Order.Ascending}, {"health_district", Order.Ascending}, {"health_facility", Order.Ascending}}),
    Final = Table.RenameColumns(Tri, {{"region", "Region"}, {"health_district", "District"}, {"health_facility", "Health facility"}})
in
    Final;

shared G_VLS = let
    P = Parametres,
    T = Transport,
    Niv = if P[Niveau] = "Region" then "region" else if P[Niveau] = "District" then "health_district"
          else if List.Count(List.Distinct(List.RemoveNulls(T[region]), Comparer.OrdinalIgnoreCase)) > 1 then "region" else "health_district",
    Mois = if P[MoisRapport] <> null then P[MoisRapport] else List.Max(T[Periode]),
    Prec = Date.AddMonths(Mois, -1),
    Z = Table.SelectRows(Table.AddColumn(T, "Zone", each Record.Field(_, Niv)), each [Zone] <> null and [Zone] <> ""),
    EstPepfar = (r) => Text.Upper(if r[site] = null then "" else r[site]) = "PEPFAR",
    G = Table.Group(Z, {"Zone"}, {
        {"Prev", (t) => List.Sum(Table.SelectRows(t, each [Periode] = Prec and EstPepfar(_))[vl_samples]), type nullable number},
        {"Cur", (t) => List.Sum(Table.SelectRows(t, each [Periode] = Mois and EstPepfar(_))[vl_samples]), type nullable number}}),
    Zeros = Table.TransformColumns(G, {{"Prev", each if _ = null then 0 else _, type number}, {"Cur", each if _ = null then 0 else _, type number}}),
    Tri = Table.Sort(Zeros, {{"Cur", Order.Descending}, {"Zone", Order.Ascending}}),
    Final = Table.FirstN(Tri, 40)
in
    Final;
